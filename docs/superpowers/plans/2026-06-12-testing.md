> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Rust unit tests for data-layer serialization logic and Playwright tests for all user-facing React behaviors, using a Tauri IPC mock injected before page load.

**Architecture:** Rust tests live in a `#[cfg(test)]` module inside `lib.rs` and test pure serde/fs logic with no Tauri runtime. Playwright tests run against the Vite dev server at `localhost:1420`; a plain-JS init script sets up `window.__TAURI_INTERNALS__` with an in-memory note store and event bus before React mounts, making all `invoke()` and `listen()` calls work without the native binary. Tests declare global type extensions in `tests/globals.d.ts`.

**Tech Stack:** Rust + serde_json + tempfile (unit tests), Playwright + TypeScript (browser tests), Vite dev server

Create `D:\Projects\scratchpad\tests\mocks\tauri-ipc.js`:

```javascript
(function () {
  // In-memory state
  let notes = [];
  let eventListeners = {};
  let callbackRegistry = {};
  let nextId = 1;
  let currentSettings = {
    theme: "dark",
    always_on_top: false,
    opacity: 1.0,
    storage_path: "",
  };

  // Tauri 2 internals: transformCallback stores a JS function and returns a numeric ID.
  // The invoke() success/error paths call these by ID.
  function transformCallback(fn, once) {
    const id = nextId++;
    callbackRegistry[id] = once
      ? function (v) {
          delete callbackRegistry[id];
          fn(v);
        }
      : fn;
    return id;
  }

  // Dispatch table for all invoke commands the app uses
  async function dispatch(cmd, args) {
    switch (cmd) {
      case "get_notes":
        return JSON.parse(JSON.stringify(notes));

      case "create_new_note": {
        const note = {
          id: crypto.randomUUID(),
          title: "Notepad Test",
          content: "",
          created_at: Date.now(),
          private: false,
        };
        notes.push(note);
        return JSON.parse(JSON.stringify(note));
      }

      case "save_note": {
        const updated = JSON.parse(JSON.stringify(args.note));
        const idx = notes.findIndex((n) => n.id === updated.id);
        if (idx >= 0) notes[idx] = updated;
        else notes.push(updated);
        return null;
      }

      case "delete_note":
        notes = notes.filter((n) => n.id !== args.noteId);
        return null;

      case "rename_note": {
        const note = notes.find((n) => n.id === args.noteId);
        if (note) note.title = args.newTitle;
        return null;
      }

      case "get_settings":
        return JSON.parse(JSON.stringify(currentSettings));

      // Tauri event system: listen() calls invoke("plugin:event|listen", { event, handler: callbackId })
      case "plugin:event|listen": {
        const { event, handler } = args;
        if (!eventListeners[event]) eventListeners[event] = [];
        eventListeners[event].push(handler); // handler is a callback ID in callbackRegistry
        return nextId++; // return an event handle (used for unlisten)
      }

      case "plugin:event|unlisten":
        return null;

      default:
        console.warn("[tauri-mock] Unhandled command:", cmd, args);
        return null;
    }
  }

  // The __TAURI_INTERNALS__ object is what @tauri-apps/api reads.
  // invoke(cmd, args, successCbId, errorCbId) — success/error are callback IDs.
  window.__TAURI_INTERNALS__ = {
    transformCallback,
    invoke: function (cmd, args, successCbId, errorCbId) {
      dispatch(cmd, args)
        .then(function (result) {
          var cb = callbackRegistry[successCbId];
          if (cb) cb(result);
        })
        .catch(function (err) {
          var cb = callbackRegistry[errorCbId];
          if (cb) cb(String(err));
        });
    },
  };

  // Test helpers — called from Playwright via page.evaluate()
  window.__TEST_EMIT__ = function (event, payload) {
    var cbIds = eventListeners[event] || [];
    cbIds.forEach(function (cbId) {
      var cb = callbackRegistry[cbId];
      if (cb) cb({ event: event, payload: payload, id: Date.now() });
    });
  };

  window.__TEST_RESET__ = function () {
    notes = [];
    eventListeners = {};
    callbackRegistry = {};
    nextId = 1;
    currentSettings = {
      theme: "dark",
      always_on_top: false,
      opacity: 1.0,
      storage_path: "",
    };
  };
})();
```
